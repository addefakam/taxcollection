     1→'use client'
     2→
     3→import { useEffect, useState, useCallback } from 'react'
     4→import { Card, CardContent, CardHeader, CardTitle } from '@/components/ui/card'
     5→import { Button } from '@/components/ui/button'
     6→import { Input } from '@/components/ui/input'
     7→import { Label } from '@/components/ui/label'
     8→import { Badge } from '@/components/ui/badge'
     9→import { Skeleton } from '@/components/ui/skeleton'
    10→import {
    11→  Select,
    12→  SelectContent,
    13→  SelectItem,
    14→  SelectTrigger,
    15→  SelectValue,
    16→} from '@/components/ui/select'
    17→import {
    18→  Dialog,
    19→  DialogContent,
    20→  DialogHeader,
    21→  DialogTitle,
    22→  DialogFooter,
    23→  DialogDescription,
    24→} from '@/components/ui/dialog'
    25→import {
    26→  Sheet,
    27→  SheetContent,
    28→  SheetHeader,
    29→  SheetTitle,
    30→  SheetDescription,
    31→  SheetFooter,
    32→} from '@/components/ui/sheet'
    33→import { useToast } from '@/hooks/use-toast'
    34→import { useRole } from './role-context'
    35→import {
    36→  TAX_CATEGORIES,
    37→  fmtDate,
    38→  fmtETB,
    39→  categoryLabel,
    40→} from './lib'
    41→import {
    42→  UserPlus,
    43→  Search,
    44→  Pencil,
    45→  Trash2,
    46→  Users,
    47→  Phone,
    48→  Mail,
    49→  MapPin,
    50→  X,
    51→  ShieldOff,
    52→  ShieldCheck,
    53→  AlertTriangle,
    54→  Landmark,
    55→  UserCircle,
    56→} from 'lucide-react'
    57→
    58→interface Woreda {
    59→  id: string
    60→  name: string
    61→}
    62→
    63→interface TaxPayer {
    64→  id: string
    65→  tin: string
    66→  fullName: string
    67→  category: string
    68→  phone?: string | null
    69→  email?: string | null
    70→  address?: string | null
    71→  woredaId?: string | null
    72→  woreda?: Woreda | null
    73→  kebele?: string | null
    74→  tradeLicenseNo?: string | null
    75→  contactPerson?: string | null
    76→  notes?: string | null
    77→  active: boolean
    78→  blocked: boolean
    79→  blockedReason?: string | null
    80→  blockedAt?: string | null
    81→  createdAt: string
    82→  _count?: { assignments: number }
    83→  assignments?: Array<{
    84→    id: string
    85→    referenceNo: string
    86→    status: string
    87→    amountDue: number
    88→    taxType: string
    89→    fiscalYear: string
    90→    period: string
    91→  }>
    92→}
    93→
    94→const emptyForm = {
    95→  tin: '',
    96→  fullName: '',
    97→  category: 'BUSINESS',
    98→  phone: '',
    99→  email: '',
   100→  address: '',
   101→  woredaId: '',
   102→  kebele: '',
   103→  tradeLicenseNo: '',
   104→  contactPerson: '',
   105→  notes: '',
   106→}
   107→
   108→export function TaxPayers() {
   109→  const { role, staffId, staffName } = useRole()
   110→  const canEdit = role === 'LEADER' || role === 'TEAM_1_ASSIGNMENT'
   111→  const isLeader = role === 'LEADER'
   112→  const { toast } = useToast()
   113→  const [list, setList] = useState<TaxPayer[]>([])
   114→  const [woredas, setWoredas] = useState<Woreda[]>([])
   115→  const [loading, setLoading] = useState(true)
   116→  const [q, setQ] = useState('')
   117→  const [category, setCategory] = useState('')
   118→  const [showBlocked, setShowBlocked] = useState(false)
   119→
   120→  const [createOpen, setCreateOpen] = useState(false)
   121→  const [editing, setEditing] = useState<TaxPayer | null>(null)
   122→  const [form, setForm] = useState(emptyForm)
   123→  const [saving, setSaving] = useState(false)
   124→
   125→  const [detail, setDetail] = useState<TaxPayer | null>(null)
   126→  const [detailLoading, setDetailLoading] = useState(false)
   127→
   128→  const load = useCallback(() => {
   129→    setLoading(true)
   130→    const params = new URLSearchParams()
   131→    if (q) params.set('q', q)
   132→    if (category) params.set('category', category)
   133→    if (!showBlocked) params.set('blocked', 'false')
   134→    fetch(`/api/taxpayers?${params}`)
   135→      .then((r) => r.json())
   136→      .then((d) => setList(d))
   137→      .finally(() => setLoading(false))
   138→  }, [q, category, showBlocked])
   139→
   140→  useEffect(() => {
   141→    const t = setTimeout(load, 250)
   142→    return () => clearTimeout(t)
   143→  }, [load])
   144→
   145→  useEffect(() => {
   146→    fetch('/api/woredas')
   147→      .then((r) => r.json())
   148→      .then((d) => setWoredas(d))
   149→  }, [])
   150→
   151→  const openCreate = () => {
   152→    setForm(emptyForm)
   153→    setEditing(null)
   154→    setCreateOpen(true)
   155→  }
   156→
   157→  const openEdit = (tp: TaxPayer) => {
   158→    setEditing(tp)
   159→    setForm({
   160→      tin: tp.tin,
   161→      fullName: tp.fullName,
   162→      category: tp.category,
   163→      phone: tp.phone || '',
   164→      email: tp.email || '',
   165→      address: tp.address || '',
   166→      woredaId: tp.woredaId || '',
   167→      kebele: tp.kebele || '',
   168→      tradeLicenseNo: tp.tradeLicenseNo || '',
   169→      contactPerson: tp.contactPerson || '',
   170→      notes: tp.notes || '',
   171→    })
   172→    setCreateOpen(true)
   173→  }
   174→
   175→  const openDetail = (tp: TaxPayer) => {
   176→    setDetail(tp)
   177→    setDetailLoading(true)
   178→    fetch(`/api/taxpayers/${tp.id}`)
   179→      .then((r) => r.json())
   180→      .then((d) => setDetail(d))
   181→      .finally(() => setDetailLoading(false))
   182→  }
   183→
   184→  const save = async () => {
   185→    if (!form.tin || !form.fullName || !form.category) {
   186→      toast({
   187→        title: 'Missing fields',
   188→        description: 'TIN, Full Name, and Category are required.',
   189→        variant: 'destructive',
   190→      })
   191→      return
   192→    }
   193→    setSaving(true)
   194→    try {
   195→      const res = await fetch(
   196→        editing ? `/api/taxpayers/${editing.id}` : '/api/taxpayers',
   197→        {
   198→          method: editing ? 'PUT' : 'POST',
   199→          headers: { 'Content-Type': 'application/json' },
   200→          body: JSON.stringify(form),
   201→        }
   202→      )
   203→      const data = await res.json()
   204→      if (!res.ok) throw new Error(data.error || 'Failed to save')
   205→      toast({
   206→        title: editing ? 'Tax payer updated' : 'Tax payer added',
   207→        description: `${data.fullName} (${data.tin})`,
   208→      })
   209→      setCreateOpen(false)
   210→      load()
   211→    } catch (e: unknown) {
   212→      const msg = e instanceof Error ? e.message : 'Failed to save'
   213→      toast({ title: 'Error', description: msg, variant: 'destructive' })
   214→    } finally {
   215→      setSaving(false)
   216→    }
   217→  }
   218→
   219→  const deactivate = async (tp: TaxPayer) => {
   220→    if (!confirm(`Deactivate tax payer ${tp.fullName}?`)) return
   221→    const res = await fetch(`/api/taxpayers/${tp.id}`, { method: 'DELETE' })
   222→    if (res.ok) {
   223→      toast({ title: 'Tax payer deactivated', description: tp.fullName })
   224→      load()
   225→    }
   226→  }
   227→
   228→  const toggleBlock = async (tp: TaxPayer) => {
   229→    if (tp.blocked) {
   230→      if (!confirm(`Unblock tax payer ${tp.fullName}?`)) return
   231→      const res = await fetch(`/api/taxpayers/${tp.id}/block`, {
   232→        method: 'PATCH',
   233→        headers: { 'Content-Type': 'application/json' },
   234→        body: JSON.stringify({ blocked: false }),
   235→      })
   236→      if (res.ok) {
   237→        toast({ title: 'Tax payer unblocked', description: tp.fullName })
   238→        load()
   239→        if (detail && detail.id === tp.id) openDetail(tp)
   240→      }
   241→    } else {
   242→      const reason = prompt(`Block tax payer ${tp.fullName}? Enter reason:`)
   243→      if (reason === null) return
   244→      const res = await fetch(`/api/taxpayers/${tp.id}/block`, {
   245→        method: 'PATCH',
   246→        headers: { 'Content-Type': 'application/json' },
   247→        body: JSON.stringify({
   248→          blocked: true,
   249→          reason,
   250→          blockedByStaffId: staffId,
   251→        }),
   252→      })
   253→      if (res.ok) {
   254→        toast({ title: 'Tax payer blocked', description: tp.fullName })
   255→        load()
   256→        if (detail && detail.id === tp.id) openDetail(tp)
   257→      }
   258→    }
   259→  }
   260→
   261→  return (
   262→    <div className="space-y-4">
   263→      {/* Toolbar */}
   264→      <div className="flex flex-wrap items-center gap-3">
   265→        <div className="relative flex-1 min-w-[200px]">
   266→          <Search className="absolute left-3 top-1/2 h-4 w-4 -translate-y-1/2 text-muted-foreground" />
   267→          <Input
   268→            placeholder="Search by TIN, name, phone, or trade license..."
   269→            value={q}
   270→            onChange={(e) => setQ(e.target.value)}
   271→            className="pl-9"
   272→          />
   273→        </div>
   274→        <Select value={category || 'ALL'} onValueChange={(v) => setCategory(v === 'ALL' ? '' : v)}>
   275→          <SelectTrigger className="w-[180px]">
   276→            <SelectValue placeholder="Category" />
   277→          </SelectTrigger>
   278→          <SelectContent>
   279→            <SelectItem value="ALL">All categories</SelectItem>
   280→            {TAX_CATEGORIES.map((c) => (
   281→              <SelectItem key={c.value} value={c.value}>
   282→                {c.label}
   283→              </SelectItem>
   284→            ))}
   285→          </SelectContent>
   286→        </Select>
   287→        <Button
   288→          variant={showBlocked ? 'default' : 'outline'}
   289→          onClick={() => setShowBlocked((v) => !v)}
   290→          className="gap-2"
   291→        >
   292→          <AlertTriangle className="h-4 w-4" />
   293→          {showBlocked ? 'Showing blocked' : 'Hide blocked'}
   294→        </Button>
   295→        {canEdit && (
   296→          <Button onClick={openCreate} className="gap-2">
   297→            <UserPlus className="h-4 w-4" /> Add Tax Payer
   298→          </Button>
   299→        )}
   300→      </div>
   301→
   302→      {/* List */}
   303→      <Card>
   304→        <CardHeader>
   305→          <CardTitle className="flex items-center gap-2 text-base">
   306→            <Users className="h-4 w-4 text-primary" />
   307→            Tax Payer Registry
   308→            <Badge variant="secondary" className="ml-1">
   309→              {loading ? '…' : list.length}
   310→            </Badge>
   311→          </CardTitle>
   312→        </CardHeader>
   313→        <CardContent className="p-0">
   314→          {loading ? (
   315→            <div className="space-y-2 p-4">
   316→              {[1, 2, 3, 4, 5].map((i) => (
   317→                <Skeleton key={i} className="h-14" />
   318→              ))}
   319→            </div>
   320→          ) : list.length === 0 ? (
   321→            <div className="p-8 text-center">
   322→              <p className="text-sm text-muted-foreground">
   323→                No tax payers found. {canEdit && 'Click "Add Tax Payer" to create the first one.'}
   324→              </p>
   325→            </div>
   326→          ) : (
   327→            <>
   328→              {/* Desktop table */}
   329→              <div className="hidden md:block overflow-x-auto">
   330→                <table className="w-full text-sm">
   331→                  <thead className="border-y bg-muted/40 text-xs uppercase tracking-wide text-muted-foreground">
   332→                    <tr>
   333→                      <th className="px-4 py-2 text-left font-medium">TIN</th>
   334→                      <th className="px-4 py-2 text-left font-medium">Name</th>
   335→                      <th className="px-4 py-2 text-left font-medium">Category</th>
   336→                      <th className="px-4 py-2 text-left font-medium">Contact</th>
   337→                      <th className="px-4 py-2 text-left font-medium">Woreda</th>
   338→                      <th className="px-4 py-2 text-center font-medium">Assignments</th>
   339→                      <th className="px-4 py-2 text-right font-medium">Actions</th>
   340→                    </tr>
   341→                  </thead>
   342→                  <tbody className="divide-y">
   343→                    {list.map((tp) => (
   344→                      <tr
   345→                        key={tp.id}
   346→                        className="hover:bg-muted/40 cursor-pointer"
   347→                        onClick={() => openDetail(tp)}
   348→                      >
   349→                        <td className="px-4 py-3 font-mono text-xs">{tp.tin}</td>
   350→                        <td className="px-4 py-3 font-medium">
   351→                          {tp.fullName}
   352→                          {tp.blocked && (
   353→                            <Badge variant="destructive" className="ml-2 text-xs">
   354→                              Blocked
   355→                            </Badge>
   356→                          )}
   357→                          {!tp.active && !tp.blocked && (
   358→                            <Badge variant="secondary" className="ml-2 text-xs">
   359→                              Inactive
   360→                            </Badge>
   361→                          )}
   362→                        </td>
   363→                        <td className="px-4 py-3">
   364→                          <Badge variant="outline">{categoryLabel(tp.category)}</Badge>
   365→                        </td>
   366→                        <td className="px-4 py-3 text-xs text-muted-foreground">
   367→                          {tp.phone || '—'}
   368→                        </td>
   369→                        <td className="px-4 py-3 text-xs text-muted-foreground">
   370→                          {tp.woreda ? `Woreda: ${tp.woreda.name}` : '—'}
   371→                        </td>
   372→                        <td className="px-4 py-3 text-center">
   373→                          <Badge variant="secondary">{tp._count?.assignments || 0}</Badge>
   374→                        </td>
   375→                        <td className="px-4 py-3 text-right" onClick={(e) => e.stopPropagation()}>
   376→                          <div className="flex justify-end gap-1">
   377→                            {isLeader && (
   378→                              <Button
   379→                                size="sm"
   380→                                variant="ghost"
   381→                                onClick={() => toggleBlock(tp)}
   382→                                title={tp.blocked ? 'Unblock' : 'Block'}
   383→                              >
   384→                                {tp.blocked ? (
   385→                                  <ShieldCheck className="h-3.5 w-3.5 text-emerald-600" />
   386→                                ) : (
   387→                                  <ShieldOff className="h-3.5 w-3.5 text-rose-600" />
   388→                                )}
   389→                              </Button>
   390→                            )}
   391→                            {canEdit && (
   392→                              <>
   393→                                <Button
   394→                                  size="sm"
   395→                                  variant="ghost"
   396→                                  onClick={() => openEdit(tp)}
   397→                                >
   398→                                  <Pencil className="h-3.5 w-3.5" />
   399→                                </Button>
   400→                                <Button
   401→                                  size="sm"
   402→                                  variant="ghost"
   403→                                  onClick={() => deactivate(tp)}
   404→                                >
   405→                                  <Trash2 className="h-3.5 w-3.5 text-destructive" />
   406→                                </Button>
   407→                              </>
   408→                            )}
   409→                          </div>
   410→                        </td>
   411→                      </tr>
   412→                    ))}
   413→                  </tbody>
   414→                </table>
   415→              </div>
   416→
   417→              {/* Mobile cards */}
   418→              <div className="md:hidden divide-y">
   419→                {list.map((tp) => (
   420→                  <button
   421→                    key={tp.id}
   422→                    onClick={() => openDetail(tp)}
   423→                    className="block w-full text-left p-4 hover:bg-muted/40"
   424→                  >
   425→                    <div className="flex items-start justify-between gap-2">
   426→                      <div className="min-w-0">
   427→                        <p className="font-medium truncate">{tp.fullName}</p>
   428→                        <p className="text-xs text-muted-foreground font-mono">{tp.tin}</p>
   429→                      </div>
   430→                      <div className="flex flex-col items-end gap-1">
   431→                        <Badge variant="outline">{categoryLabel(tp.category)}</Badge>
   432→                        {tp.blocked && (
   433→                          <Badge variant="destructive" className="text-[10px]">
   434→                            Blocked
   435→                          </Badge>
   436→                        )}
   437→                      </div>
   438→                    </div>
   439→                    <div className="mt-2 flex items-center justify-between text-xs text-muted-foreground">
   440→                      <span className="flex items-center gap-1">
   441→                        <Phone className="h-3 w-3" /> {tp.phone || '—'}
   442→                      </span>
   443→                      {tp.woreda && <Badge variant="secondary">Woreda: {tp.woreda.name}</Badge>}
   444→                    </div>
   445→                  </button>
   446→                ))}
   447→              </div>
   448→            </>
   449→          )}
   450→        </CardContent>
   451→      </Card>
   452→
   453→      {/* Create / Edit dialog */}
   454→      <Dialog open={createOpen} onOpenChange={setCreateOpen}>
   455→        <DialogContent className="max-w-2xl max-h-[90vh] overflow-y-auto">
   456→          <DialogHeader>
   457→            <DialogTitle>{editing ? 'Edit Tax Payer' : 'Add New Tax Payer'}</DialogTitle>
   458→            <DialogDescription>
   459→              {editing
   460→                ? 'Update the tax payer details below.'
   461→                : 'Register a new tax payer in the Bishoftu Finance registry.'}
   462→            </DialogDescription>
   463→          </DialogHeader>
   464→
   465→          <div className="grid gap-4 sm:grid-cols-2">
   466→            <div className="space-y-1.5">
   467→              <Label>TIN (Tax ID) *</Label>
   468→              <Input
   469→                value={form.tin}
   470→                onChange={(e) => setForm({ ...form, tin: e.target.value })}
   471→                placeholder="BT-0011-2026"
   472→              />
   473→            </div>
   474→            <div className="space-y-1.5">
   475→              <Label>Category *</Label>
   476→              <Select
   477→                value={form.category}
   478→                onValueChange={(v) => setForm({ ...form, category: v })}
   479→              >
   480→                <SelectTrigger>
   481→                  <SelectValue />
   482→                </SelectTrigger>
   483→                <SelectContent>
   484→                  {TAX_CATEGORIES.map((c) => (
   485→                    <SelectItem key={c.value} value={c.value}>
   486→                      {c.label}
   487→                    </SelectItem>
   488→                  ))}
   489→                </SelectContent>
   490→              </Select>
   491→            </div>
   492→            <div className="space-y-1.5 sm:col-span-2">
   493→              <Label>Full Name / Business Name *</Label>
   494→              <Input
   495→                value={form.fullName}
   496→                onChange={(e) => setForm({ ...form, fullName: e.target.value })}
   497→                placeholder="e.g. Bekele Tadesse"
   498→              />
   499→            </div>
   500→            <div className="space-y-1.5">
   501→              <Label>Phone</Label>
   502→              <Input
   503→                value={form.phone}
   504→                onChange={(e) => setForm({ ...form, phone: e.target.value })}
   505→                placeholder="+2519..."
   506→              />
   507→            </div>
   508→            <div className="space-y-1.5">
   509→              <Label>Email</Label>
   510→              <Input
   511→                value={form.email}
   512→                onChange={(e) => setForm({ ...form, email: e.target.value })}
   513→                placeholder="email@example.com"
   514→              />
   515→            </div>
   516→            <div className="space-y-1.5">
   517→              <Label>Woreda (sub-city)</Label>
   518→              <Select
   519→                value={form.woredaId || 'NONE'}
   520→                onValueChange={(v) => setForm({ ...form, woredaId: v === 'NONE' ? '' : v })}
   521→              >
   522→                <SelectTrigger>
   523→                  <SelectValue placeholder="Select woreda..." />
   524→                </SelectTrigger>
   525→                <SelectContent>
   526→                  <SelectItem value="NONE">— None —</SelectItem>
   527→                  {woredas.map((w) => (
   528→                    <SelectItem key={w.id} value={w.id}>
   529→                      {w.name}
   530→                    </SelectItem>
   531→                  ))}
   532→                </SelectContent>
   533→              </Select>
   534→            </div>
   535→            <div className="space-y-1.5">
   536→              <Label>Kebele</Label>
   537→              <Input
   538→                value={form.kebele}
   539→                onChange={(e) => setForm({ ...form, kebele: e.target.value })}
   540→                placeholder="03"
   541→              />
   542→            </div>
   543→            <div className="space-y-1.5">
   544→              <Label>Trade License No.</Label>
   545→              <Input
   546→                value={form.tradeLicenseNo}
   547→                onChange={(e) => setForm({ ...form, tradeLicenseNo: e.target.value })}
   548→                placeholder="BL-XXX-YYYY"
   549→              />
   550→            </div>
   551→            <div className="space-y-1.5">
   552→              <Label>Contact Person</Label>
   553→              <Input
   554→                value={form.contactPerson}
   555→                onChange={(e) => setForm({ ...form, contactPerson: e.target.value })}
   556→                placeholder="e.g. Ato Bekele"
   557→              />
   558→            </div>
   559→            <div className="space-y-1.5 sm:col-span-2">
   560→              <Label>Address</Label>
   561→              <Input
   562→                value={form.address}
   563→                onChange={(e) => setForm({ ...form, address: e.target.value })}
   564→                placeholder="Street, landmark, etc."
   565→              />
   566→            </div>
   567→            <div className="space-y-1.5 sm:col-span-2">
   568→              <Label>Notes</Label>
   569→              <Input
   570→                value={form.notes}
   571→                onChange={(e) => setForm({ ...form, notes: e.target.value })}
   572→                placeholder="Any internal note"
   573→              />
   574→            </div>
   575→          </div>
   576→
   577→          <DialogFooter>
   578→            <Button variant="outline" onClick={() => setCreateOpen(false)} disabled={saving}>
   579→              Cancel
   580→            </Button>
   581→            <Button onClick={save} disabled={saving}>
   582→              {saving ? 'Saving…' : editing ? 'Save Changes' : 'Add Tax Payer'}
   583→            </Button>
   584→          </DialogFooter>
   585→        </DialogContent>
   586→      </Dialog>
   587→
   588→      {/* Detail sheet */}
   589→      <Sheet open={!!detail} onOpenChange={(o) => !o && setDetail(null)}>

   603→                    <Badge variant="secondary">Inactive</Badge>

   623→                  </div>

   706→                </div>

   708→              <SheetFooter className="flex-col gap-2 sm:flex-row">

   717→                  >
   718→                    <Pencil className="h-3.5 w-3.5" /> Edit

   721→                {isLeader && (

   723→                    variant={detail.blocked ? 'default' : 'outline'}

   738→                <Button variant="ghost" onClick={() => setDetail(null)}>
   739→                  <X className="h-4 w-4" /> Close
   740→                </Button>

   742→            </>
   743→          )}
   744→        </SheetContent>
   745→      </Sheet>
   746→    </div>
   747→  )
   748→}
